     1	/**
     2	 * Antum People — the privacy notice the server serves.
     3	 *
     4	 * ---------------------------------------------------------------------------
     5	 * ONE DEFINITION OF "THE NOTICE"
     6	 * ---------------------------------------------------------------------------
     7	 * An acknowledgement record's evidential value rests on one thing: the version it names must be
     8	 * the version of the notice a client could actually have been shown. So the route that serves the
     9	 * notice and the route that records an acknowledgement read the SAME file through this module.
    10	 * Two independent resolutions could disagree, and a record stating a version the served notice does
    11	 * not carry is precisely the defect the acknowledgement work exists to remove.
    12	 *
    13	 * The resolution order below is the one the template route already used before this module existed
    14	 * (the host-shared templates directory first, this repository's own `templates/` as the fallback).
    15	 * It is kept rather than chosen because changing what the product serves is a decision with its own
    16	 * blast radius, not a side effect of this one — see the note on divergence below.
    17	 *
    18	 * ---------------------------------------------------------------------------
    19	 * A KNOWN DIVERGENCE, RECORDED NOT FIXED
    20	 * ---------------------------------------------------------------------------
    21	 * At the time of writing the two copies differ in content (not in version): the copy this module
    22	 * resolves first still carries the words "statutory retention" in its retention table, and the
    23	 * repository's copy does not — that qualifier was removed as unsourced in an earlier change that
    24	 * landed in the repository only. So the notice a client can download today is the older text.
    25	 * That is a finding for the owner of the notice, not something this module should paper over by
    26	 * silently preferring the repository's copy: which copy the product serves is a deliberate choice,
    27	 * and until it is made this module reads what the product actually serves and says so here.
    28	 */
    29	
    30	const fs = require('fs');
    31	const path = require('path');
    32	
    33	/** The template name the notice is served under (`GET /api/compliance/templates/:name/:employeeId`). */
    34	const NOTICE_TEMPLATE = 'privacy-notice';
    35	
    36	/** Where the product's shared templates live, and where this repository keeps its own. */
    37	const SHARED_TEMPLATES_DIR = '/home/team/shared/templates';
    38	const REPO_TEMPLATES_DIR = path.join(__dirname, '..', 'templates');
    39	
    40	/**
    41	 * The absolute path of the notice the server serves. The first existing candidate wins, so a host
    42	 * without the shared directory still serves the repository's copy rather than failing.
    43	 */
    44	function noticePath() {
    45	  const shared = path.join(SHARED_TEMPLATES_DIR, `${NOTICE_TEMPLATE}.md`);
    46	  if (fs.existsSync(shared)) return shared;
    47	  return path.join(REPO_TEMPLATES_DIR, `${NOTICE_TEMPLATE}.md`);
    48	}
    49	
    50	/** The notice's markdown, exactly as it is served. */
    51	function readNotice() {
    52	  return fs.readFileSync(noticePath(), 'utf8');
    53	}
    54	
    55	/**
    56	 * The notice's own version string, read from its version header line:
    57	 *
    58	 *     > **Version / الإصدار:** v2.2 (2026-09-17) — owner legal details applied
    59	 *
    60	 * Never a literal in code. The rule that keeps a record true is that changing the notice's content
    61	 * means bumping this line — a content change that did not bump it is how a record ends up
    62	 * describing a notice nobody can produce. So the record reads whatever the served file says, and
    63	 * the `Arabic: الإصدار` half of the header is matched structurally rather than by its characters.
    64	 *
    65	 * @throws {Error} when the served file carries no version header at all — better a loud failure
    66	 *                 than a record written against an unknown version.
    67	 */
    68	function noticeVersion() {
    69	  const match = readNotice().match(/^\s*>\s*\*\*Version[^*]*\*\*:?\s*(v[0-9][0-9A-Za-z.\-]*)/m);
    70	  if (!match) {
    71	    throw new Error(
    72	      `No version header in ${noticePath()}: expected a line like '> **Version / ...:** v2.2'`
    73	    );
    74	  }
    75	  return match[1];
    76	}
    77	
    78	module.exports = {
    79	  NOTICE_TEMPLATE,
    80	  noticePath,
    81	  readNotice,
    82	  noticeVersion,
    83	};
